# ---- 服务层：RED ----
server_rps          sum(rate(http_requests_total{job="enrolld",route="${route}"}[${window}]))
server_error_rate   (sum(rate(http_requests_total{job="enrolld",route="${route}",code=~"5.."}[${window}])) or vector(0)) / sum(rate(http_requests_total{job="enrolld",route="${route}"}[${window}]))
server_p99_ms       1000 * histogram_quantile(0.99, sum by (le) (rate(http_request_duration_seconds_bucket{job="enrolld",route="${route}"}[${window}])))
server_mean_ms      1000 * sum(rate(http_request_duration_seconds_sum{job="enrolld",route="${route}"}[${window}])) / sum(rate(http_request_duration_seconds_count{job="enrolld",route="${route}"}[${window}]))
